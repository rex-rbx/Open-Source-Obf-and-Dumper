local proxy = newproxy(true)
local metatable = getmetatable(proxy)

metatable.__tostring = function()
	return string.rep("A", 1e8)
end

metatable.__index = function()
	return proxy
end

clone = function()
	return proxy
end

local t3 = {}
local function insert()
	table.insert(t3, clone(t3))
end

while t3 do task.spawn(insert) end
